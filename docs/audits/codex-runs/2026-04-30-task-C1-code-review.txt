    22	  /** Wave 2 起追加 runtime context（kernel session / theme / etc.） */
    23	}
    24	
    25	/**
    26	 * BlockUI：presentational 层 —— React 组件，绑定到已注册的 core。
    27	 * 同一 core 可挂多个 UI（uiId 区分），Adopter 可写自己的 UI 替代 ui-default。
    28	 */
    29	export interface BlockUIDefinition<TSchema extends ZodTypeAny = ZodTypeAny> {
    30	  /** 必须指向已 registerCore 的 core */
    31	  readonly coreName: string;
    32	  /** UI 实现 id：'default' / 'minimal' / 用户自定义 */
    33	  readonly uiId: string;
    34	  readonly EditorView: ComponentType<BlockViewProps<TSchema>>;
    35	  readonly RenderView: ComponentType<BlockViewProps<TSchema>>;
    36	}
    37	
    38	export function defineCore<T extends ZodTypeAny>(
    39	  def: BlockCoreDefinition<T>,
    40	): BlockCoreDefinition<T> {
    41	  return def;
    42	}
    43	
    44	export function defineUI<T extends ZodTypeAny>(
    45	  def: BlockUIDefinition<T>,
    46	): BlockUIDefinition<T> {
    47	  return def;
    48	}
    49	
    50	/**
    51	 * BlockRegistry：双层注册表（ADR-0003）。
    52	 * registerCore：挂业务定义；registerUI：挂视觉实现。
    53	 * 同 core 多 UI 场景：getUI(name) 不带 uiId 时取首个注册的（约定为 'default'）。
    54	 */
    55	export class BlockRegistry {
    56	  readonly #cores = new Map<string, BlockCoreDefinition>();
    57	  readonly #uis = new Map<string, BlockUIDefinition[]>();
    58	
    59	  registerCore(core: BlockCoreDefinition): void {
    60	    if (this.#cores.has(core.name)) {
    61	      throw new Error(`Duplicate core name: ${core.name}`);
    62	    }
    63	    this.#cores.set(core.name, core);
    64	  }
    65	
    66	  registerUI(ui: BlockUIDefinition): void {
    67	    if (!this.#cores.has(ui.coreName)) {
    68	      throw new Error(`Unknown core for UI registration: ${ui.coreName}`);
    69	    }
    70	    const existing = this.#uis.get(ui.coreName) ?? [];
    71	    if (existing.some((u) => u.uiId === ui.uiId)) {
    72	      throw new Error(
    73	        `Duplicate UI registration: core=${ui.coreName} uiId=${ui.uiId}`,
    74	      );
    75	    }
    76	    existing.push(ui);
    77	    this.#uis.set(ui.coreName, existing);
    78	  }
    79	
    80	  getCore(name: string): BlockCoreDefinition | undefined {
    81	    return this.#cores.get(name);
    82	  }
    83	
    84	  listCores(): readonly BlockCoreDefinition[] {
    85	    return [...this.#cores.values()];
    86	  }
    87	
    88	  /** 不传 uiId 时返回首个（约定为 'default'）；传了精确匹配 */
    89	  getUI(coreName: string, uiId?: string): BlockUIDefinition | undefined {
    90	    const list = this.#uis.get(coreName);
    91	    if (!list || list.length === 0) return undefined;
    92	    if (uiId === undefined) return list[0];
    93	    return list.find((u) => u.uiId === uiId);
    94	  }
    95	
    96	  listUIs(coreName: string): readonly BlockUIDefinition[] {
    97	    return [...(this.#uis.get(coreName) ?? [])];
    98	  }
    99	}

    12	- `proseExtensions` — Tiptap 扩展数组，提供全部 markdown 行为

    14	## Invariants
    15	
    16	- `BlockKind` 二分不可破坏：新 block 必须明确归属 `prose` 或 `component`
    17	- Prose blocks 零自写代码——任何看似需要新 prose block 的场景应通过组合 `proseExtensions` 内现有扩展或追加单条 Tiptap 扩展实现，不再加 prose-kind block
    18	- `BlockCoreDefinition.mdxComponent` 必须 PascalCase，且与 MDX 文件 import 中使用的名字一致
    19	- Core 与 UI 物理分离（ADR-0003）：core 不允许 import 任何 React/Tiptap 视觉 API；UI 必须 import core（不允许 inline 重复 schema）
    20	- 同 core 多 UI 时 `getUI(name)` 取首个注册：约定首个 uiId 为 `'default'`；adopter 在 register 顺序上需谨慎
    21	- **Defensive copy**：`listCores()` 与 `listUIs(coreName)` 每次返回 fresh array。Caller 修改返回值不影响 registry 内部 state。Implementation: `src/registry.ts:84` (`listCores`) + `src/registry.ts:96` (`listUIs`)；regression: `src/__tests__/registry.test.ts` "listCores returns a defensive copy" + "listUIs returns a defensive copy"
    22	- **Idempotent register**：`registerCore` 同 `name` 第二次调用 throws `Duplicate core name: <name>`；`registerUI` 同 `(coreName, uiId)` 第二次 throws `Duplicate UI registration: core=<coreName> uiId=<uiId>`。避免 silent override；实现见 `src/registry.ts:60-63` + `src/registry.ts:71-75`
    23	- **Schema strictness**：每个 `BlockCoreDefinition.propsSchema` 必须是 `z.object(...).strict()`；嵌套 `ZodObject` 同样必须 `.strict()`（[ADR-0006](../../docs/decisions/ADR-0006-asymmetry-audit-checklist.md) item #3 trip-hazard：`.strict()` 不传播）。当前类型仅约束到 `ZodTypeAny`（type-narrow 到 `ZodObject` 待 Wave 3）；Wave 2 行为约定 + reviewer + RFC.md §1 显式提醒。未 `.strict()` 不会 throw，但 `propsSchema.parse(input)` 会 silently strip unknown keys，引发 contract drift
    24	- **Serialize / parse hook ownership**：MDX serialize/parse 由各 `block-*` 包的 `core/` own（ADR-0003 D1+D2），不在 `block-foundation` 注册。`block-foundation` 仅 own `BlockCoreDefinition` 形状（含 `mdxComponent` 字符串）；`mdx-bridge` 通过 `mdxJsxFlowElement.name` 字符串路由到对应 block 的 export（命名约定 verb-as-prefix：`serialize<BlockName>` / `parse<BlockName>`，与 mdx-bridge 现有 `mdxToTiptap` / `tiptapToMdx` 同 style）。详见 RFC.md §"5. Serialize / parse hook ownership"

     2	
     3	> **Audience**: `simple-block-eng` / `render-block-eng` / `viz-block-eng` writing
     4	> `block-callout` / `block-code` / `block-image` / `block-math` / `block-pdf` /
     5	> `block-jupyter` / `block-nn-viz` / `block-agent-flow` cores; `ux-ui-lead`
     6	> writing the matching `ui-default/` halves.
     7	>
     8	> **Authority**: [ADR-0008](../../docs/decisions/ADR-0008-wave-2-entry-policies.md)
     9	> D2 freezes the registration interface as of 2026-04-30 (commit `b3581bc`).
    10	> This RFC is the consumer-facing walkthrough; the public API surface itself
    11	> lives in [`CONTRACT.md`](./CONTRACT.md).
    12	
    13	This RFC is **not** a spec — `CONTRACT.md` is. It is a step-by-step recipe
    14	showing how the first real consumer (`block-callout`, Task C1/C2) wires its
    15	core + UI into a `BlockRegistry`. Subsequent `block-*` PRs follow the same
    16	shape; cloned `block-code` / `block-image` cores (Tasks C3-C5b) inherit
    17	this structure verbatim.
    18	
    19	## 1. core/ side
    20	
    21	`core/` is the headless half (ADR-0003). No React, no Tiptap; just a Zod
    22	schema + identifying metadata. Place it at `packages/<block>/src/core/`
    23	and export through the package's `core` subpath.
    24	
    25	```typescript
    26	// packages/block-callout/src/core/core-definition.ts
    27	import { defineCore, type BlockCoreDefinition } from '@skb/block-foundation';
    28	import { z } from 'zod';
    29	
    30	const propsSchema = z
    31	  .object({
    32	    variant: z.enum(['note', 'tip', 'warning', 'danger']),
    33	    title: z.string().optional(),
    34	  })
    35	  .strict();
    36	
    37	export const calloutCore: BlockCoreDefinition<typeof propsSchema> = defineCore({
    38	  name: 'callout',
    39	  kind: 'component',
    40	  propsSchema,
    41	  mdxComponent: 'Callout',
    42	});
    43	```
    44	
    45	Notes that map directly to `BlockCoreDefinition` fields in
    46	[`src/registry.ts`](./src/registry.ts):
    47	
    48	- `name`: kebab-case, globally unique. Used as the registry key — duplicate
    49	  registration throws `Duplicate core name: <name>` at runtime.
    50	- `kind`: `'prose' | 'component'`. Wave 2 blocks are all `'component'`;
    51	  `'prose'` is reserved for behaviors covered by `proseExtensions` and is
    52	  closed to new entries (CONTRACT invariant: "Prose blocks 零自写代码").
    53	- `propsSchema`: must be a `z.object(...).strict()`; every nested
    54	  `ZodObject` must independently call `.strict()`. See
    55	  [`CONTRACT.md` § Invariants — Schema strictness](./CONTRACT.md#invariants)
    56	  + [ADR-0006](../../docs/decisions/ADR-0006-asymmetry-audit-checklist.md) item #3
    57	  (`.strict()` does not propagate into nested objects). The current type
    58	  signature `BlockCoreDefinition<TSchema extends ZodTypeAny>` is wider than
    59	  `ZodObject`; type-narrowing is deferred to Wave 3, so for Wave 2 this is
    60	  a behavior contract enforced by RFC + reviewer + integration tests, not
    61	  by TypeScript. Skipping `.strict()` does not throw — `propsSchema.parse`
    62	  silently strips unknown keys, which is exactly the contract-drift class
    63	  this rule prevents.
    64	- `mdxComponent`: PascalCase, must match the JSX tag used in `.mdx` files
    65	  exactly (the MDX bridge uses this string when serializing back). For
    66	  `block-callout`, MDX content reads `<Callout variant="note">…</Callout>`
    67	  so this field is `'Callout'`.
    68	
    69	`defineCore` is a no-op identity helper that exists purely to give TypeScript
    70	inference a hand — the returned object satisfies `BlockCoreDefinition<T>`
    71	with `T` inferred from `propsSchema`. Equivalent to writing the type
    72	annotation manually. `defineUI` (used in §2) is the symmetric helper for
    73	`BlockUIDefinition<T>`; both helpers exist for inference convenience only,
    74	so consumers can write `defineUI({ ... })` without manually spelling
    75	`<typeof calloutCore.propsSchema>`.
    76	
    77	## 2. ui-default/ side
    78	
    79	`ui-default/` is the presentational half. It imports the core (never inline-
    80	redefines its schema — see ADR-0006 item #4: "single-authority schema → audit
    81	every consumer for inline duplicates"), then attaches React components for
    82	the editor and renderer.
    83	
    84	```typescript
    85	// packages/block-callout/src/ui-default/index.ts
    86	import { defineUI, type BlockUIDefinition } from '@skb/block-foundation';
    87	import { calloutCore } from '../core/core-definition';
    88	import { CalloutEditorView } from './editor-view';
    89	import { CalloutRenderView } from './render-view';
    90	
    91	export const calloutUIDefault: BlockUIDefinition<
    92	  typeof calloutCore.propsSchema
    93	> = defineUI({
    94	  coreName: calloutCore.name, // must reference an already-registered core
    95	  uiId: 'default', // convention: first-registered UI is 'default'
    96	  EditorView: CalloutEditorView,
    97	  RenderView: CalloutRenderView,
    98	});
    99	```
   100	
   101	The two view components share the `BlockViewProps<TSchema>` shape from
   102	[`src/registry.ts`](./src/registry.ts):
   103	
   104	```typescript
   105	// packages/block-callout/src/ui-default/editor-view.tsx
   106	import type { ComponentType } from 'react';
   107	import type { BlockViewProps } from '@skb/block-foundation';
   108	import { calloutCore } from '../core/core-definition';
   109	
   110	export const CalloutEditorView: ComponentType<
   111	  BlockViewProps<typeof calloutCore.propsSchema>
   112	> = ({ props, content }) => {
   113	  // props.variant is z-inferred to 'note' | 'tip' | 'warning' | 'danger'
   114	  // props.title is z-inferred to string | undefined
   115	  // content is the optional Tiptap inner content as a string
   116	  return (
   117	    <aside data-variant={props.variant}>
   118	      {props.title ? <strong>{props.title}</strong> : null}
   119	      <div>{content}</div>
   120	    </aside>
   121	  );
   122	};
   123	```
   124	
   125	`RenderView` follows the same shape but renders for static output (Astro
   126	build / SSR path) rather than the editor. UX/visual conventions for the
   127	component itself (tokens, spacing, accessibility) are owned by `ux-ui-lead`
   128	via `packages/design-tokens/CONTRACT.md`; this RFC fixes only the
   129	registration plumbing.
   130	
   131	### Wiring at boot
   132	
   133	`registerCore` must precede `registerUI` for the same `coreName`; otherwise
   134	`registerUI` throws `Unknown core for UI registration: <coreName>`. The
   135	boot path (Wave 3 `editor-shell` / `apps/site` integration) does:
   136	
   137	```typescript
   138	import { BlockRegistry } from '@skb/block-foundation';
   139	import { calloutCore } from '@skb/block-callout/core';
   140	import { calloutUIDefault } from '@skb/block-callout/ui-default';
   141	
   142	const registry = new BlockRegistry();
   143	registry.registerCore(calloutCore);
   144	registry.registerUI(calloutUIDefault);
   145	
   146	// later, render-time lookup:
   147	const ui = registry.getUI('callout'); // returns calloutUIDefault (uiId='default')
   148	const ui2 = registry.getUI('callout', 'minimal'); // exact match by uiId
   149	```
   150	
   151	When multiple UIs are registered for one core, `getUI(name)` without a
   152	`uiId` returns the **first registered** (CONTRACT invariant — convention:
   153	register `'default'` first). `getUI(name, uiId)` does exact match.
   154	
   155	## 3. Consumer-side test template
   156	
   157	Each `block-*` package ships an integration test that wires its own core +
   158	UI into a fresh registry and verifies the round-trip. This is the minimum
   159	surface; package-specific tests (props validation, schema edge cases, view
   160	rendering) live alongside.
   161	
   162	```typescript
   163	// packages/block-callout/src/__tests__/registry-integration.test.ts

    44	## Decision
    45	
    46	Adopt an **8-point asymmetry-audit checklist** as a mandatory part of `code-reviewer` (Codex 5.3-spark, default cheap) and `pr-gate` (Codex 5.5, deep) review prompts. Reviewers MUST work through every applicable item before declaring PASS.
    47	
    48	### The 8 items
    49	
    50	1. **Field/attribute audit**: After adding a field to a data shape (Zod schema, Pydantic model, TS interface, mdast attrs, etc.), audit every comparator / equality / serialization / canonicalization function that handles this shape. (Wave 1 evidence: instance #4 — `marksEqual` link branch missed `title` when serialize emitted it.)
    51	
    52	2. **Status code audit**: After adding a status code to one HTTP handler (or, generally, an exit/error code to one branch), audit every other handler emitting the same code. RFC-mandated headers (RFC 7235 `WWW-Authenticate` on 401, RFC 7231 `Allow` on 405) attach to **the status code itself**, not the framework code path that emitted it. (Wave 1 evidence: instance #2 — custom `SkbError` 401 path missed `WWW-Authenticate`.)
    53	
    54	3. **Strictness audit**: After adding `.strict()` (Zod) / `extra='forbid'` (Pydantic) at one level of an object schema, audit every nested object/sub-schema. The strictness setting **does not propagate**; nested objects must opt in independently. (Wave 1 evidence: instance #3 — top-level `.strict()` shipped while nested `frontmatterSchema` still silently stripped unknown keys.)
    55	
    56	4. **Consumer schema audit**: After defining a schema in a single-authority package (e.g. `@skb/content-types/frontmatterSchema`), audit every consumer for inline duplicates or inline redefinitions that should `import` instead. The "deferred swap" pattern — "I'll inline now and import later" — is a contract break that compounds over time. (Wave 1 evidence: instance #6 part 2 — apps/site/content.config.ts inlined `z.object(...)` rather than importing `frontmatterSchema`.)
    57	
    58	5. **Algorithm + runtime constant replication audit**: When an algorithm or runtime constant is replicated outside its authority (e.g. inline FOUC scripts, login challenge derivation, pre-hydration boot logic that cannot import from packages), audit every consumer-side replica for byte-equivalent reproduction. The replica must register a **regression test** that asserts equivalence against the authority across a representative input corpus. (Wave 1 evidence: instance #6 part 1 — FOUC IIFE truthy-coerce vs `getInitialTheme()` strict-whitelist; remediated with `apps/site/src/__tests__/fouc-script.test.ts` 16-row corpus.)
    59	
    60	6. **Sister-file documentation audit**: After updating one CONTRACT.md (or any document that describes a code surface), audit ALL related CONTRACT.md files in the package family for sister-file drift. Two CONTRACT.md files that describe two halves of the same contract must use **identical terminology** for the shared identifiers (function names, schema names, refine predicate names). (Wave 1 evidence: instance #5 — `editor-commands/CONTRACT.md` updated to cite `editBlockInputSchema` while `agent-tools/CONTRACT.md` still referred to the old shape.)
    61	
    62	7. **Exception-scope equivalence audit**: When replicating an algorithm + its exception handling across files, audit BOTH the happy-path semantics AND the `try/catch` scope between authority and replica. Narrow vs wide `try/catch` can produce divergent fall-through behavior even with identical happy-path logic. Specifically: when `localStorage` / `sessionStorage` / `IndexedDB` / `matchMedia` / any browser API in the algorithm path can throw, the `try/catch` MUST wrap ONLY the throwing call, with the fall-through branch (matchMedia / system default / DOM apply) running AFTER the catch handler — not inside it. (Wave 1 evidence: instance #7 — FOUC IIFE wide-scope try/catch swallowed matchMedia exceptions; remediated with narrow try/catch + 2 storage-throws regression tests.)
    63	
    64	8. **Authority-document → generated/consumed surface audit**: When updating an authority document (ADR, spec, `agent-contract.md`, runbook, CONVENTIONS) **OR a dependency authority (`package.json`)** that mandates behavior in downstream consumers, audit every consumer-side replica/generated artifact for synchronized update **AND staged-in-the-git-index in the same bundle**. Specifically: an ADR section that says "X file MUST be updated to do Y" requires the same commit to update X and provide a way to verify the synchronization. Generated artifacts (e.g. `docs/review-checklist.md` and `.claude/agents/*.md` derived from `agent-contract.md`; **`pnpm-lock.yaml` derived from `package.json`** — see sub-form C below) MUST be regenerated via the documented codegen step (`pnpm generate:configs` / `pnpm install`) AND **explicitly `git add`'d** so the git **index** (not just the working tree) contains the new content; runbooks (`docs/runbooks/team-operations.md`) MUST be cross-linked; Codex/Claude subagent prompt files MUST embed or reference the new rule.

    67	   ```bash

    69	   # 2. Run codegen
    70	   pnpm generate:configs
    71	   # 3. Stage regenerated outputs (CRITICAL — see Pitfall below)
    72	   git add <regenerated/path/foo.md> <regenerated/path/bar.md>
    73	   # 4. Verify index reflects content (NOT just working tree):
    74	   git diff --cached --stat -- <regenerated paths>
    75	   #    → expects non-empty diff with the expected line counts
    76	   # 5. Verify idempotence: re-run codegen + confirm working tree matches index:
    77	   pnpm generate:configs
    78	   git status --short -- <regenerated paths>
    79	   #    → expects column-1 'M' (staged) + column-2 blank (working tree clean)
    80	   ```
    81	
    82	   **Pitfall (this is the load-bearing step)**: `git status --short` shows two columns: column 1 is index-vs-HEAD; column 2 is working-tree-vs-index. After running the generator without `git add`, you see `' M'` (space + M) — **working tree differs from index, but index still matches HEAD**. `git commit` only commits the index. Skipping `git add` means the regenerated content is in your working tree but NOT in the next commit. The bundle would land authority + generator-code changes WITHOUT the regenerated outputs, leaving consumers (Wave 2 reviewers reading `.claude/agents/*.md`) at pre-update content. Use `git diff --cached` (not `git status`) as the canonical staging-verification command — `--cached` shows index-vs-HEAD only, which is what `git commit` will produce.
    83	
    84	   **Pitfall extension — `pnpm-lock.yaml` is a generated artifact too (sub-form C below)**: when the authority is `package.json` (any dependency add / remove / version bump / `@types/*` shift), `pnpm install` regenerates `pnpm-lock.yaml` as a deterministic derived artifact. Skipping `git add pnpm-lock.yaml` produces the same class of failure as skipping `git add .claude/agents/*.md`: working tree has the synced lockfile but the **index** still holds HEAD's stale lockfile, so the bundle ships `package.json` ahead of its lockfile. CI runs `pnpm install --frozen-lockfile` from the bundled state, sees the stale lockfile disagreeing with the new `package.json`, and fails install. The same `git diff --cached --stat -- pnpm-lock.yaml` canonical verification applies. (Wave 1 erratum commit `e15ec36`; codified as sub-form C below.)
    85	
    86	   Generated-from-authority files carry their own header comment ("GENERATED FROM agent-contract.md — DO NOT EDIT BY HAND") OR YAML frontmatter declaring authority source — both forms are acceptable. `pnpm-lock.yaml` lacks an explicit header (the format is fixed by pnpm) but is implicitly declared by `pnpm install`'s contract — its authority is `package.json` + the workspace `pnpm-workspace.yaml`.
    87	
    88	   **Wave 1 evidence**: **instance #8** — ADR-0006 v1 commit codified reviewer-prompt mandates but did not update `agent-contract.md` / regenerate `.claude/agents/code-reviewer.md` / regenerate `.claude/agents/pr-gate.md` / cross-link `team-operations.md` / regenerate `docs/review-checklist.md`; pr-gate caught the dogfood failure during R1 review of the close-ceremony bundle, forcing a v2 fix. **instance #12 (sub-form A — pre-regeneration omission)** — v5 bundle updated authority files (`agent-contract.md`, generator code, `team-operations.md` cross-link) but had not run the generator at all post-edit; the 3 regenerated outputs were stale at HEAD content; pr-gate caught this during R2 review, forcing a v6 fix. **instance #12 (sub-form B — staging-vs-working-tree confusion)** — v6 bundle ran the generator (so working tree had new content) but did NOT `git add` the regenerated outputs; the git **index** still held HEAD blob hashes for the 3 generated files; orchestrator confused `git status --short` "M" prefix (working-tree-vs-index) with index-vs-HEAD; pr-gate caught this via `git ls-files --stage` blob-hash comparison during R3 review, forcing the v7 fix that explicitly stages with `git add` + verifies via `git diff --cached`. **instance #12 (sub-form C — lockfile is a generated-from-authority artifact)** — Wave 1 close ceremony commit `e15ec36` ("fix(deps): commit pnpm-lock.yaml for @tailwindcss/typography (lockfile drift)"): the design-tokens erratum (commit `fdc86a8`) added the `@tailwindcss/typography` dependency to `package.json` but did not stage the regenerated `pnpm-lock.yaml`; CI install used the pre-update lockfile, drifted against the new `package.json`, and failed `pnpm install --frozen-lockfile`. Resolution: commit `e15ec36` staged the lockfile separately. Lesson codified: when the authority is `package.json` (any dependency add / remove / version bump), `pnpm-lock.yaml` is the consumer-side generated artifact and must ride the same bundle — the class equivalence to `.claude/agents/*.md` regeneration is exact (deterministic derive + git-index staging required). The four sub-forms of the same item #8 class — pre-regeneration omission, post-regeneration commit-staging confusion, conceptually-correct-but-operationally-incomplete fix, AND lockfile-as-generated-artifact — together motivate the explicit operational sequence and `git diff --cached` canonical verification command codified above.
    89

    43	### D1：dead-dep policy = Option A（tighten）
    44
