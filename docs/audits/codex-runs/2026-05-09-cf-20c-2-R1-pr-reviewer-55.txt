   118	/**
   119	 * Walk the editor doc and return one IdentifiedBlock per `.skb-block-nodeview`
   120	 * NodeView. Block id = ProseMirror node `pos` as string. Used at
   121	 * drag-start to populate the snapshot; never re-called during
   122	 * drag-over (snapshot semantics per ADR-0017 D6 lift mode).
   123	 */
   124	function snapshotBlocks(editor: Editor): SerializedBlock[] {
   125	  const blocks: SerializedBlock[] = [];
   126	  editor.state.doc.descendants((node, pos) => {
   127	    const isBlockNode = node.type.spec['group']?.toString().includes('block') ?? false;
   128	    if (!isBlockNode || !node.attrs) return undefined;
   129	    // Filter to NodeView blocks only (skip prose paragraphs etc.).
   130	    const colAttr: unknown = node.attrs['col'];
   131	    if (typeof colAttr !== 'number') return undefined;
   132	    const rowAttr: unknown = node.attrs['row'];
   133	    blocks.push({
   134	      id: String(pos),
   135	      pmPos: pos,
   136	      nodeName: node.type.name,
   137	      col: colAttr,
   138	      ...(typeof rowAttr === 'number' && { row: rowAttr }),
   139	      colSpan: node.attrs['colSpan'] as number,
   140	      rowSpan: (node.attrs['rowSpan'] as number | 'auto') ?? 1,
   141	    });
   142	    return false; // Don't descend into block nodes.
   143	  });
   144	  return blocks;
   145	}
   146	
   147	/**
   148	 * Measure DOM bounding rects for every snapshot block. Selector pattern
   149	 * `.skb-block-nodeview` is the editor-mount path (cf-19); we look up
   150	 * the wrapper for each snapshot block by walking the editor's DOM.
   151	 */
   152	function measureBlockRects(
   153	  editor: Editor,
   154	  blocks: readonly SerializedBlock[],
   155	): Map<string, DOMRectReadOnly> {
   156	  const rects = new Map<string, DOMRectReadOnly>();
   157	  const editorEl = editor.view.dom;
   158	  for (const block of blocks) {
   159	    const node = editor.view.nodeDOM(block.pmPos);
   160	    const el = node instanceof HTMLElement ? node : null;
   161	    if (el) {
   162	      rects.set(block.id, el.getBoundingClientRect());
   163	    } else {
   164	      // Fallback: scan for matching kind under editor DOM root.
   165	      const fallback = editorEl.querySelector(
   166	        `.skb-block-nodeview[data-skb-block-kind="${block.nodeName}"]`,
   167	      );
   168	      if (fallback instanceof HTMLElement) {
   169	        rects.set(block.id, fallback.getBoundingClientRect());
   170	      }
   171	    }
   172	  }
   173	  return rects;
   174	}
   175	
   176	/**
   177	 * Translate snapshot id → live pmPos at drop-time. The snapshot id IS
   178	 * the pmPos string at drag-start; positions may shift if an external
   179	 * transaction mutated the doc during drag (rare but possible). We
   180	 * walk the live doc and match by (nodeName + original pmPos) as a
   181	 * best-effort; if a block can't be located, skip its attr update.
   182	 */
   183	function liveBlockPositions(
   184	  editor: Editor,
   185	  snapshot: readonly SerializedBlock[],
   186	): Map<string, { pos: number; nodeName: string }> {
   187	  const map = new Map<string, { pos: number; nodeName: string }>();
   188	  const liveBlocks: Array<{ pos: number; nodeName: string }> = [];
   189	  editor.state.doc.descendants((node, pos) => {
   190	    if (typeof node.attrs?.['col'] !== 'number') return undefined;
   191	    liveBlocks.push({ pos, nodeName: node.type.name });
   192	    return false;
   193	  });
   194	  // Same-order match (sample-blocks fixtures all colSpan=12; block
   195	  // count is stable during a single drag transaction). Future
   196	  // cf-20c-3+ cross-doc-mutation case adds a stable id field per
   197	  // ADR-0016 D2 amendment.
   198	  for (let i = 0; i < snapshot.length && i < liveBlocks.length; i++) {
   199	    const snapBlock = snapshot[i];
   200	    const liveBlock = liveBlocks[i];
   201	    if (snapBlock && liveBlock && snapBlock.nodeName === liveBlock.nodeName) {
   202	      map.set(snapBlock.id, { pos: liveBlock.pos, nodeName: liveBlock.nodeName });
   203	    }
   204	  }
   205	  return map;
   206	}

   330	## Decisions
   331	
   332	### D1 — HTML5 native DnD (Q7 spike result: works inside ProseMirror NodeView contenteditable=false gutter)
   333	
   334	**Trigger**: Q7 open question from the cf-20 scoping report —
   335	"verify HTML5 native DnD works inside ProseMirror NodeView with
   336	`contenteditable=false` host element. If blocked: pivot to pointer
   337	events."
   338	
   339	**Q7 spike method**: throwaway Playwright spec (`__cf20c2-q7-spike.spec.ts`,
   340	deleted post-verdict) injected a `<button draggable=true>` into the
   341	`.skb-block-nodeview__gutter` of the live `/notes/sample-blocks/edit`
   342	route, registered dragstart/dragover/drop listeners, dispatched
   343	synthetic `DragEvent` instances, and inspected the events that fired.
   344	
   345	**Q7 result**: HTML5 DnD works. Spike output:
   346	```
   347	{
   348	  "events": [
   349	    { "kind": "button:dragstart", "defaultPrevented": false, "dataTransfer": "" },
   350	    { "kind": "pm:dragover", "target": "tiptap ProseMirror", "defaultPrevented": true, "dataTransfer": "q7-spike-payload" },
   351	    { "kind": "pm:drop", "target": "tiptap ProseMirror ProseMirror-focused", "defaultPrevented": true, "dataTransfer": "q7-spike-payload" }
   352	  ],
   353	  "dragoverFireCount": 1,
   354	  "pmHasOwnDragstart": false
   355	}
   356	```
   357	
   358	The dragstart fired on the button; `dataTransfer.setData()` succeeded;
   359	dragover and drop received our payload (`q7-spike-payload`) on the
   360	`.ProseMirror` target. ProseMirror did NOT cancel the drag (no
   361	`pmHasOwnDragstart` interception). The `defaultPrevented: true` on
   362	dragover and drop is from OUR handler calling `preventDefault()` (required
   363	so the browser accepts drop on the editor).
   364	
   365	**Decision**: cf-20c-2 ships HTML5 native DnD. Saves ~150 LOC of
   366	pointer-events scaffolding (manual ghost following cursor, manual
   367	hit-test on mousemove, etc.) and the native API integrates cleanly
   368	with the existing `drag-ghost` / `drop-pulse` / `outline-overlay`
   369	primitives.
   370	
   371	**Defense-in-depth**: the button calls `event.stopPropagation()` on
   372	both dragstart and dragend. Q7 spike showed PM doesn't intercept by
   373	default, but a future Tiptap upgrade or an extension that adds a
   374	NodeSelection drag handler could compete. The stopPropagation
   375	explicitly walls cf-20c-2's lifecycle from PM's.
   376	
   377	### D2 — Block ID sourcing: ProseMirror node `pos` (Path A) NOT stable UUID attrs (Path B)
   378	
   379	**Open question** from cf-20 scoping report — block-id sourcing for
   380	the algebra layer. Two paths:
   381	
   382	- **Path A**: use ProseMirror node `pos` as a string. `pos` is
   383	  stable WITHIN a single doc transaction; the pipeline snapshots
   384	  positions at drag-start and only re-walks the doc at drop.
   385	  Tiptap suspends input handling during a focused drag, so pos
   386	  values won't shift unless an external transaction interleaves —
   387	  which doesn't happen in the cf-20c-2 single-user single-session
   388	  scope (per ADR-0016 D12 explicit assumption).
   389	- **Path B**: add a stable UUID `id` attr to every block schema, generate
   390	  at NodeView mount, persist through MDX serialize / parse. Requires:
   391	  (i) Zod schema additions in 8 block packages, (ii) mdx-bridge
   392	  serialize/parse round-trip, (iii) migration of existing notes
   393	  (or hash-based defaults), (iv) sister-doc updates across 11+ files.
   394	
   395	**Decision**: Path A. Three reasons:
   396	1. cf-20c-2 is the integration PR; adding a schema change here
   397	   would mix two layers of risk (drag wire + persistent ID
   398	   migration) in one PR. Per the orchestrator's "ship the wire
   399	   then iterate" framing.
   400	2. Path A satisfies ADR-0016 D12's single-user single-session
   401	   assumption — stable IDs are only required for collaborative /
   402	   multi-session scenarios that are explicit Phase 2+ scope.
   403	3. Future cf-20c-3+ that introduces palette → grid drag (new block
   404	   insertion) will naturally need a stable client-generated UUID
   405	   anyway (the new block has no pre-existing pos); Path B can land
   406	   then with a clean rationale.
   407	
   408	Trade-off accepted: a doc transaction WHILE a drag is active
   409	(extremely rare; would require an extension or async data-load
   410	trigger to dispatch a transaction during user drag) could
   411	invalidate `pos` mappings and the pipeline's `liveBlockPositions()`
   412	fallback (same-order match by nodeName) would mis-route attr
   413	updates. Documented in `use-drag-drop-pipeline.ts` JSDoc.
   414	
   415	### D3 — React Context bridges per-block button → editor mount pipeline (NOT prop-drilling through factory args)
   416	
   417	`BlockNodeView` is rendered inside Tiptap's `ReactNodeViewRenderer`
   418	which mounts each NodeView in a SEPARATE React subtree (Tiptap
   419	creates new React roots for each NodeView; props flow only via the
   420	Tiptap node attrs). Prop-drilling drag callbacks through
   421	`makeBlockNodeView({registry, onDragStart, onDragEnd, ...})`
   422	factory args was rejected because:
   423	- The factory is invoked once per block kind at `wireRegistry`
   424	  call time (NOT per-block); callbacks would have to be stable
   425	  across all 8 block-kind instances.
   426	- The pipeline is mounted at `EditorShellMount.tsx` which lives
   427	  in apps/site, while `wireRegistry` lives in editor-shell — the
   428	  factory args boundary would force apps/site to construct the
   429	  registry, which would require duplicating the editor-shell
   430	  registration glue at apps/site.
   431	- React context is the canonical bridge for "presentational
   432	  child callbacks routing to a distant lifecycle owner".
   433	
   434	cf-20c-2 introduces `DragDropContext` + `DragDropProvider`. The
   435	provider is mounted at `EditorShellMount.tsx` wrapping
   436	`<GridContainer>`; the per-block button consumes via `useContext(DragDropContext)`.
   437	Default `null` value = degraded mode (button renders, callbacks
   438	no-op without throwing) — covers test mounts that don't provide
   439	the pipeline.
   440	
   441	### D4 — Ghost kind hardcoded to `'markdown'` at cf-20c-2 (BlockKind sourcing deferred to follow-up PR)
   442	
   443	`<DragGhost kind="markdown" mode="move">` is mounted with a
   444	hardcoded kind because:
   445	- cf-20c-2's pipeline tracks `sourceBlockId` (a string) but not
   446	  the source's BlockKind. To source the kind, the pipeline would
   447	  need to look up the snapshot block by id at active-render time
   448	  AND map the kind name to a `GhostKind` (`'canvas' | 'runnable'
   449	  | 'image' | 'markdown'`).
   450	- The ghost's role is purely visual (cursor follower); the kind
   451	  affects color but not behavior. `markdown` (neutral gray per
   452	  ADR-0017 D10) is a safe default.
   453	- cf-20d (resize) will need the same kind lookup for resize
   454	  handles AND introduces a `BlockKindForGhost` mapping in
   455	  `block-foundation`; deferring to cf-20d centralizes the
   456	  kind-mapping logic.
   457	
   458	Future PR: source `kind = blockKindToGhostKind(snapshot[sourceBlockId].kind)`
   459	once the mapping helper lands.
   460	
   461	### D5 — `useDragDropPipeline` owns its own internal `layoutReducer` state (NOT shared with note-save-adapter)
   462	
   463	The hook calls `useReducer(layoutReducer, INITIAL_LAYOUT_STATE)` to
   464	manage `epoch + snapshot + baseline`. This state lives entirely
   465	within the hook; it does NOT participate in the cf-20c-1+ algebra's
   466	external `baseline: GridSnapshotIdentified` parameter (which is
   467	computed fresh from the snapshot at drag-start).
   468	
   469	Rationale: at cf-20c-2 the `layoutState` is forward-compat scaffolding
   470	— the cf-20c-1 algebra is pure (input → output), the
   471	`setNodeMarkup` calls are the actual persistent mutations (Tiptap's
   472	own undo/redo handles them), and the layoutReducer's epoch is unused
   473	downstream. cf-20d (resize) will dispatch `drag-end-success` with
   474	its own `mutation` parameter; cf-20e (kebab-menu undo) might need
   475	the epoch for "undo last drag" semantics. cf-20c-2 surfaces
   476	`layoutState` in the hook return so future consumers can read it
   477	without re-architecting.
   478	
   479	### D6 — Pipeline rolls back via `drag-end-mode-none` when no edge match at drop (vs accept "empty" mode default)
   480	
   481	When `tiebreak()` returns null (cursor not on any edge rect at drop
   482	moment), the pipeline dispatches `drag-end-mode-none` per ADR-0017
   483	D1 line 53 ("cursor on host non-edge"). The cf-20c-1 algebra
   484	supports `mode='empty'` for grid-empty-area drops, but cf-20c-2
   485	doesn't compute `emptyTarget` because:
   486	- Sample-blocks fixtures are all `colSpan=12` so there's no
   487	  visible empty grid surface to drop into during the cf-20c-2 demo.
   488	- `empty` mode requires computing `(col, row)` from cursor position
   489	  via grid math; that's a non-trivial layer that warrants its own
   490	  PR (cf-20c-3 candidate; not in current 9-PR sequence — out of
   491	  cf-20c-2 scope per orchestrator's "drag handle UI + DnD wire"
   492	  framing).
   493	
   494	cf-20c-2 ships drag-between-existing-blocks only. `mode-none`
   495	rollback is the safe fallback.
   496	
   497	### D7 — c4-3 spec re-targeted to `/notes/sample-blocks/edit` (was `/notes/sample-mdx-note/edit`)
   498	
   499	The pre-cf-20c-2 c4-3 spec asserted the standalone DragHandle on a
   500	prose-only route. Post-cf-20c-2 the standalone DragHandle is removed
   501	and the per-block handles only exist where there are NodeViews
   502	(component blocks). Pure prose routes have ZERO per-block handles —
   503	which is correct (prose paragraphs don't get drag handles per the
   504	v2 contract; only component blocks do).
   505	
   506	The c4-3 spec is updated to target sample-blocks/edit (where 14
   507	NodeViews exist) and assert per-block handle structure + dragstart
   508	overlay mount. Pre-cf-20c-2 assertions on the standalone DragHandle
   509	are removed entirely (no replacement; the standalone is deprecated).
